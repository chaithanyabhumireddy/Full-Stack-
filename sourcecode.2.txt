        </div>

            <label>Username</label>

            <label>Password</label>
            <input type="password" name="password" required />

            <button type="submit">Sign In</button>
        </form>

        <p>Don't have an account? <a th:href="@{/register}">Register</a></p>

        <div th:if="${param.error}">Invalid username or password</div>
        <div th:if="${param.logout}">Logged out successfully</div>
        <div th:if="${param.registered}">Registration successful</div>

</div>
</body>
</html>
